
# Enable Rewrite Engine
RewriteEngine On
RewriteBase /

# 1. Force HTTPS and WWW prefix
# This solves the domain mismatch by ensuring everyone lands on www.asistir-srl.com.ar
RewriteCond %{HTTPS} off [OR]
RewriteCond %{HTTP_HOST} !^www\. [NC]
RewriteRule ^(.*)$ https://www.asistir-srl.com.ar/$1 [L,R=301,NE]

# 2. Redirect legacy .html files to clean React routes
# This prevents the "Not Secure" error shown in your screenshot
RewriteRule ^index\.html$ / [R=301,L]
RewriteRule ^servicios\.html$ /servicios [R=301,L]
RewriteRule ^quienes-somos\.html$ /quienes-somos [R=301,L]
RewriteRule ^contacto\.html$ /contacto [R=301,L]
RewriteRule ^examenes\.html$ /examenes [R=301,L]
RewriteRule ^ausentismo\.html$ /ausentismo [R=301,L]

# 3. Prevent direct access to .tsx files (Internal Security)
RewriteRule \.(tsx|ts|json)$ - [F,L]

# 4. BrowserRouter Support
# If the requested path is not a physical file (like an image), send it to index.html
RewriteCond %{REQUEST_FILENAME} -f [OR]
RewriteCond %{REQUEST_FILENAME} -d
RewriteRule ^ - [L]
RewriteRule ^ index.html [L]

# 5. Security Headers
<IfModule mod_headers.c>
    Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains; preload"
    Header always set X-Content-Type-Options "nosniff"
    Header always set X-Frame-Options "SAMEORIGIN"
    Header always set X-XSS-Protection "1; mode=block"
</IfModule>
